# Block definitions. Each [name] section is one block; its ID is its position in this file.
#
# Face textures (file names in textures/); later lines override earlier ones:
#   all, side (the four horizontal faces), top, bottom,
#   positive_x, negative_x, positive_y, negative_y, positive_z, negative_z
# Optional:
#   opaque = false                 non-opaque block (glass, leaves): its ID gets bit 9 (512+) and it is meshed
#                                  separately from the binary mesher; texture alpha < 0.5 is cut out
#   tint = <r> <g> <b>             color multiplier for every face at every LOD (e.g. per-liquid color of the shared
#                                  grayscale fluid texture)
#   frame_time = <seconds>         seconds per frame of animated textures (default 0.1). A texture 16 px wide and
#                                  16 * N tall is a strip of N animation frames, top first
#   fluid = ocean                  the ocean: fills empty cells below sea level, an infinite source that only
#                                  placing a block removes; drawn translucent (needs opaque = false)
#   hardness = <number>            Minecraft's destroy time (default 0: breaks instantly in survival)
#   requires_tool = true           needs a tool to drop in Minecraft; breaking by hand is 3.3x slower
#   random_rotation = <faces...>   rotate these faces' textures by a per-face hash to break up tiling
#   lod_<faces> = <texture>        texture used instead on coarse LOD chunks
#   lod_tint = <r> <g> <b>         color multiplier applied with the lod_ texture
#   far_tint = <r> <g> <b>         color multiplier faded in with distance around LOD 3 (where a tree is one leaf block)
